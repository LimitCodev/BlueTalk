# Sube el paquete de voces (empaquetar.py) al CDN: voces/<voz>/v1/.
#
# Uso:  scripts/aws/publicar_voces.sh [Lupe]              # simulación
#       DE_VERDAD=1 scripts/aws/publicar_voces.sh Lupe    # sube de verdad
#

# o corregido va en v2 (y la app pasa a pedir v2).
source "$(dirname "$0")/comun.sh"

VOZ="${1:-Lupe}"
PAQUETE="$RAIZ/build/voces/$VOZ/paquete/v1"
for f in voces.bin indice.json; do
  [[ -f "$PAQUETE/$f" ]] || { echo "Falta $PAQUETE/$f: corre empaquetar.py" >&2; exit 1; }
done
[[ "${DE_VERDAD:-0}" == "1" ]] && requiere aws

BUCKET="${BLUETALK_BUCKET:-$(salida_pila Bucket)}"
DESTINO="s3://$BUCKET/voces/$VOZ/v1"
INMUTABLE="public, max-age=31536000, immutable"
# Sin --delete: el bucket también guarda el sitio legal.
ejecutar aws s3 cp "$PAQUETE/voces.bin" "$DESTINO/voces.bin" \
  --cache-control "$INMUTABLE" --content-type application/octet-stream
ejecutar aws s3 cp "$PAQUETE/indice.json" "$DESTINO/indice.json" \
  --cache-control "$INMUTABLE" --content-type application/json
echo "Paquete en: $(salida_pila UrlBase)/voces/$VOZ/v1/indice.json"

`VozChannel.kt` (completo):

```kotlin
package com.limitcodev.bluetalk

import android.app.Activity
import android.media.AudioAttributes
import android.media.MediaPlayer
import io.flutter.plugin.common.BinaryMessenger
import io.flutter.plugin.common.MethodCall
import io.flutter.plugin.common.MethodChannel
import java.io.File
import java.io.FileInputStream

/**
 * Edición Play: reproduce un tramo de voces.bin (paquete de voces
 * naturales). Solo bajo un toque: Flutter llama a `reproducir` desde
 * speakText. Un audio nuevo corta el anterior.
 */
class VozChannel(
    private val activity: Activity,
    messenger: BinaryMessenger,
) : MethodChannel.MethodCallHandler {

    private val canal = MethodChannel(messenger, "bluetalk/voz")
    private var reproductor: MediaPlayer? = null
    private val carpeta: File
        get() = File(activity.filesDir, "voces").apply { mkdirs() }

    init {
        canal.setMethodCallHandler(this)
    }

    override fun onMethodCall(call: MethodCall, result: MethodChannel.Result) {
        when (call.method) {
            "carpeta" -> result.success(carpeta.absolutePath)
            "reproducir" -> result.success(reproducir(call))
            "parar" -> result.success(parar())
            else -> result.notImplemented()
        }
    }

    private fun reproducir(call: MethodCall): Boolean {
        val ruta = call.argument<String>("ruta") ?: return false
        val inicio = call.argument<Number>("inicio")?.toLong() ?: return false
        val largo = call.argument<Number>("largo")?.toLong() ?: return false
        val archivo = File(ruta).canonicalFile
        if (!archivo.path.startsWith(carpeta.canonicalPath + File.separator)) return false
        parar()
        return try {
            val mp = MediaPlayer()
            mp.setAudioAttributes(
                AudioAttributes.Builder()
                    .setUsage(AudioAttributes.USAGE_MEDIA)
                    .setContentType(AudioAttributes.CONTENT_TYPE_SPEECH)
                    .build(),
            )
            FileInputStream(archivo).use { mp.setDataSource(it.fd, inicio, largo) }
            mp.setOnCompletionListener { it.release(); if (reproductor === it) reproductor = null }
            mp.prepare() // tramo local y pequeño: preparar en el hilo principal tarda poco
            mp.start()
            reproductor = mp
            true
        } catch (e: Exception) {
            false
        }
    }

    private fun parar(): Boolean {
        reproductor?.run { try { stop() } catch (_: IllegalStateException) {}; release() }
        reproductor = null
        return true
    }

    fun dispose() {
        parar()
        canal.setMethodCallHandler(null)
    }
}
```

*shortcut: `prepare()` en el hilo principal; pasar a `prepareAsync()` si un tramo tarda más de ~50 ms en un teléfono de gama baja.*

1. GET `$urlVoces/voces/Lupe/v1/indice.json`.

/// TtsSpeaker que prueba primero la voz natural descargada (Plan 5) y,
/// si no hay, usa la del sistema. Nunca deja un toque en silencio.
class VozNatural implements TtsSpeaker {
  VozNatural(this._sistema, {required PaqueteVoz? Function() paquete,
      required bool Function() permitido,
      CanalNativo canal = const CanalNativo('bluetalk/voz')});

  @override
  Future<void> speak(String text) async {
    final p = permitido() ? paquete() : null;
    final t = p?.tramo(text);
    if (t != null &&
        await _canal.orden('reproducir', {'ruta': p!.ruta, 'inicio': t.$1, 'largo': t.$2})) {
      return;
    }
    await _sistema.speak(text);
  }

  @override
  Future<void> stop() async {
    await _canal.orden('parar');
    await _sistema.stop();
  }
  // setRate, setPitch y hasSpanishVoice: delegan en _sistema.
}
